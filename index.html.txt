    <script>
        const { createApp } = Vue;

        createApp({
            data() {
                return {
                    abaAtiva: 'cargas',
                    config: {
                        tensaoFN: 127,
                        tempAmbiente: 30,
                        qtdAgrupamento: 2,
                        quedaMaxPermitida: 4
                    },
                    novoComodo: {
                        nome: '',
                        tipo: 'geral',
                        area: null,
                        perimetro: null,
                        tueNome: '',
                        tuePotencia: null
                    },
                    comodos: []
                }
            },
            computed: {
                totalIluminacaoVA() {
                    return this.comodos.reduce((acc, c) => acc + c.iluminacaoVA, 0);
                },
                totalTugVA() {
                    return this.comodos.reduce((acc, c) => acc + c.tugTotalVA, 0);
                },
                totalTueW() {
                    return this.comodos.reduce((acc, c) => acc + (c.tuePotencia || 0), 0);
                },
                fatorTemp() {
                    if (this.config.tempAmbiente === 35) return 0.94;
                    if (this.config.tempAmbiente === 40) return 0.87;
                    return 1.00;
                },
                fatorAgrupamento() {
                    if (this.config.qtdAgrupamento === 2) return 0.80;
                    if (this.config.qtdAgrupamento === 3) return 0.70;
                    if (this.config.qtdAgrupamento === 4) return 0.65;
                    return 1.00;
                },
                circuitosCalculados() {
                    const lista = [];
                    let num = 1;

                    // Iluminação
                    const potIlum = this.totalIluminacaoVA;
                    if (potIlum > 0) {
                        const ib = potIlum / this.config.tensaoFN;
                        const ibCorr = ib / (this.fatorTemp * this.fatorAgrupamento);
                        const secao = this.dimensionarCabo(ibCorr, 1.5);
                        const disj = this.dimensionarDisjuntor(ibCorr, secao.iz);
                        lista.push({
                            numero: num++,
                            nome: 'Iluminação Social e Serviços',
                            tipo: 'Iluminação',
                            tensao: this.config.tensaoFN,
                            potencia: potIlum,
                            ib: ib,
                            ibCorrigida: ibCorr,
                            secaoFase: secao.bitola,
                            secaoTerra: secao.bitola,
                            disjuntor: disj
                        });
                    }

                    // TUGs Serviço
                    const comodosServico = this.comodos.filter(c => c.tipo === 'servico');
                    const potTugServico = comodosServico.reduce((acc, c) => acc + c.tugTotalVA, 0);
                    if (potTugServico > 0) {
                        const ib = potTugServico / this.config.tensaoFN;
                        const ibCorr = ib / (this.fatorTemp * this.fatorAgrupamento);
                        const secao = this.dimensionarCabo(ibCorr, 2.5);
                        const disj = this.dimensionarDisjuntor(ibCorr, secao.iz);
                        lista.push({
                            numero: num++,
                            nome: 'TUGs Cozinha e Área de Serviço',
                            tipo: 'TUG',
                            tensao: this.config.tensaoFN,
                            potencia: potTugServico,
                            ib: ib,
                            ibCorrigida: ibCorr,
                            secaoFase: secao.bitola,
                            secaoTerra: secao.bitola,
                            disjuntor: disj
                        });
                    }

                    // TUGs Gerais
                    const comodosGerais = this.comodos.filter(c => c.tipo !== 'servico');
                    const potTugGeral = comodosGerais.reduce((acc, c) => acc + c.tugTotalVA, 0);
                    if (potTugGeral > 0) {
                        const ib = potTugGeral / this.config.tensaoFN;
                        const ibCorr = ib / (this.fatorTemp * this.fatorAgrupamento);
                        const secao = this.dimensionarCabo(ibCorr, 2.5);
                        const disj = this.dimensionarDisjuntor(ibCorr, secao.iz);
                        lista.push({
                            numero: num++,
                            nome: 'TUGs Salas e Dormitórios',
                            tipo: 'TUG',
                            tensao: this.config.tensaoFN,
                            potencia: potTugGeral,
                            ib: ib,
                            ibCorrigida: ibCorr,
                            secaoFase: secao.bitola,
                            secaoTerra: secao.bitola,
                            disjuntor: disj
                        });
                    }

                    // TUEs
                    this.comodos.forEach(c => {
                        if (c.tuePotencia > 0) {
                            const tensaoTue = c.tuePotencia > 3000 ? (this.config.tensaoFN === 127 ? 220 : this.config.tensaoFN) : this.config.tensaoFN;
                            const ib = c.tuePotencia / tensaoTue;
                            const ibCorr = ib / (this.fatorTemp * 1.00);
                            const secao = this.dimensionarCabo(ibCorr, 2.5);
                            const disj = this.dimensionarDisjuntor(ibCorr, secao.iz);
                            lista.push({
                                numero: num++,
                                nome: `TUE - ${c.tueNome} (${c.nome})`,
                                tipo: 'TUE',
                                tensao: tensaoTue,
                                potencia: c.tuePotencia,
                                ib: ib,
                                ibCorrigida: ibCorr,
                                secaoFase: secao.bitola,
                                secaoTerra: secao.bitola,
                                disjuntor: disj
                            });
                        }
                    });

                    return lista;
                },
                listaMateriais() {
                    const cabos = { '1.5': 0, '2.5': 0, '4.0': 0, '6.0': 0, '10.0': 0 };

                    this.circuitosCalculados.forEach(c => {
                        const metragem = 40;
                        cabos[c.secaoFase] = (cabos[c.secaoFase] || 0) + (metragem * 3);
                    });

                    const qtdTugTotal = this.comodos.reduce((acc, c) => acc + c.qtdTug, 0);
                    const qtdTueTotal = this.comodos.filter(c => c.tuePotencia > 0).length;

                    const potTotalVA = (this.totalIluminacaoVA + this.totalTugVA) * 0.5 + (this.totalTueW * 0.8);
                    const correnteGeral = potTotalVA / 220;
                    const disjuntorGeralVal = correnteGeral > 50 ? 63 : (correnteGeral > 40 ? 50 : 40);

                    return {
                        cabos: cabos,
                        disjuntorGeral: disjuntorGeralVal,
                        idrGeral: disjuntorGeralVal,
                        qdtModulos: Math.max(12, this.circuitosCalculados.length + 6),
                        tug10A: qtdTugTotal,
                        tue20A: qtdTueTotal,
                        interruptores: Math.ceil(this.comodos.length * 1.2),
                        eletroduto34: Math.ceil(this.comodos.length * 16),
                        caixas4x2: qtdTugTotal + Math.ceil(this.comodos.length * 1.2)
                    };
                }
            },
            methods: {
                calcularCargasNBR(c) {
                    let ilumVA = 100;
                    if (c.area >= 6) {
                        const dif = c.area - 6;
                        ilumVA += Math.floor(dif / 4) * 60;
                    }

                    let qtdTug = 1;
                    let tugVA = 100;

                    if (c.tipo === 'servico') {
                        qtdTug = Math.max(1, Math.ceil(c.perimetro / 3.5));
                        tugVA = qtdTug <= 3 ? (qtdTug * 600) : (3 * 600 + (qtdTug - 3) * 100);
                    } else {
                        if (c.area >= 6) qtdTug = Math.max(1, Math.ceil(c.perimetro / 5.0));
                        tugVA = qtdTug * 100;
                    }

                    return { ilumVA, qtdTug, tugVA };
                },
                adicionarComodo() {
                    const res = this.calcularCargasNBR(this.novoComodo);
                    this.comodos.push({
                        nome: this.novoComodo.nome,
                        tipo: this.novoComodo.tipo,
                        area: this.novoComodo.area,
                        perimetro: this.novoComodo.perimetro,
                        iluminacaoVA: res.ilumVA,
                        qtdTug: res.qtdTug,
                        tugTotalVA: res.tugVA,
                        tueNome: this.novoComodo.tueNome || '',
                        tuePotencia: this.novoComodo.tuePotencia || 0
                    });

                    this.novoComodo.nome = '';
                    this.novoComodo.area = null;
                    this.novoComodo.perimetro = null;
                    this.novoComodo.tueNome = '';
                    this.novoComodo.tuePotencia = null;
                },
                carregarExemplo() {

                    ];
                },
                removerComodo(index) {
                    this.comodos.splice(index, 1);
                },
                dimensionarCabo(ibCorr, minBitola) {
                    const tabelaB1 = [
                        { bitola: 1.5, iz: 17.5 },
                        { bitola: 2.5, iz: 24.0 },
                        { bitola: 4.0, iz: 32.0 },
                        { bitola: 6.0, iz: 41.0 },
                        { bitola: 10.0, iz: 57.0 }
                    ];

                    for (let c of tabelaB1) {
                        if (c.bitola >= minBitola && c.iz >= ibCorr) return c;
                    }
                    return { bitola: 10.0, iz: 57.0 };
                },
                dimensionarDisjuntor(ibCorr, iz) {
                    const disjPadrao = [10, 16, 20, 25, 32, 40, 50, 63];
                    for (let d of disjPadrao) {
                        if (d >= ibCorr && d <= iz) return d;
                    }
                    return 50;
                },
                imprimirRelatorio() {
                    window.print();
                }
            }
        }).mount('#app');
